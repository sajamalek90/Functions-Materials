Function Fundamentals

Function Intro And Basic Usage

Functions are reusable blocks of code that perform specific tasks.

// Function declaration
function greet(name) {
    console.log(`Hello, ${name}!`);
}

// Function call
greet('Alice'); // Hello, Alice!

// Function expression
const square = function(x) {
    return x * x;
};
console.log(square(5)); // 25

Function Advanced Examples

Functions can be used in more complex scenarios.

// Function to calculate age
function calculateAge(birthYear) {
    return new Date().getFullYear() - birthYear;
}

// Function with multiple parameters
function createUser(name, age, isAdmin = false) {
    return {
        username: name,
        userAge: age,
        admin: isAdmin
    };
}

// Immediately Invoked Function Expression (IIFE)
(function() {
    console.log('This runs immediately');
})();

Function Return Statement And Use Cases

The return statement specifies the value to be returned.

// Simple return
function add(a, b) {
    return a + b;
}

// Early return
function checkAge(age) {
    if (age < 18) {
        return 'Minor';
    }
    return 'Adult';
}

// Returning multiple values (as object)
function getDimensions() {
    return {
        width: 100,
        height: 200
    };
}

// No return (returns undefined)
function noReturn() {
    console.log('No return value');
}

Advanced Function Concepts

Function Default Parameters

Default parameters provide fallback values when arguments are not provided.

function createMessage(name = 'Guest', message = 'Welcome!') {
    return `${name}, ${message}`;
}

console.log(createMessage()); // Guest, Welcome!
console.log(createMessage('Alice')); // Alice, Welcome!
console.log(createMessage('Bob', 'Hello!')); // Bob, Hello!

// Default parameters can be expressions
function createId(prefix = 'usr', random = Math.random().toString(36).slice(2)) {
    return `${prefix}_${random}`;
}

Function Rest Parameters

Rest parameters allow functions to accept any number of arguments as an array.

function sum(...numbers) {
    return numbers.reduce((total, num) => total + num, 0);
}

console.log(sum(1, 2, 3)); // 6
console.log(sum(10, 20, 30, 40)); // 100

// Rest parameter must be last
function logItems(title, ...items) {
    console.log(title);
    items.forEach(item => console.log(item));
}

Function Ultimate Practice

Comprehensive example combining multiple function concepts.

function createShoppingCart(discount = 0) {
    const items = [];
    
    return {
        addItem: function(name, price) {
            items.push({ name, price });
        },
        removeItem: function(name) {
            const index = items.findIndex(item => item.name === name);
            if (index !== -1) items.splice(index, 1);
        },
        calculateTotal: function() {
            const subtotal = items.reduce((sum, item) => sum + item.price, 0);
            return subtotal * (1 - discount);
        },
        getItems: function() {
            return [...items]; // Return copy
        }
    };
}

const cart = createShoppingCart(0.1); // 10% discount
cart.addItem('Shirt', 25);
cart.addItem('Pants', 40);
console.log(cart.calculateTotal()); // 58.5

Anonymous Function And Practice

Anonymous functions are functions without a name, often used as callbacks.

// As callback
const numbers = [1, 2, 3];
numbers.forEach(function(number) {
    console.log(number * 2);
});

// As IIFE
(function() {
    console.log('This runs immediately');
})();

// Stored in variable
const double = function(x) { return x * 2; };
console.log(double(5)); // 10

Return Nested Function

Functions can return other functions, creating closures.

function createMultiplier(factor) {
    return function(number) {
        return number * factor;
    };
}

const double = createMultiplier(2);
const triple = createMultiplier(3);

console.log(double(5)); // 10
console.log(triple(5)); // 15

// Practical example - counter
function createCounter() {
    let count = 0;
    return {
        increment: function() { count++; },
        decrement: function() { count--; },
        getCount: function() { return count; }
    };
}

Arrow Function Syntax

Arrow functions provide a concise syntax and handle this differently.

// Basic syntax
const add = (a, b) => a + b;

// Single parameter (parentheses optional)
const square = x => x * x;

// Multiple parameters
const greet = (name, age) => `Hello ${name}, you are ${age} years old`;

// Multi-line body
const calculate = (a, b) => {
    const sum = a + b;
    const product = a * b;
    return { sum, product };
};

// No parameters
const random = () => Math.random();

Scope Concepts

Scope – Global And Local

Scope determines where variables are accessible.

// Global scope
const globalVar = 'I am global';

function testScope() {
    // Local/function scope
    const localVar = 'I am local';
    console.log(globalVar); // Accessible
    console.log(localVar); // Accessible
}

testScope();
console.log(globalVar); // Accessible
console.log(localVar); // Error: localVar is not defined

Scope – Block

let and const are block-scoped (within curly braces).

if (true) {
    const blockVar = 'I am block scoped';
    let anotherVar = 'Me too';
    console.log(blockVar); // Accessible
}

console.log(blockVar); // Error: blockVar is not defined

for (let i = 0; i < 3; i++) {
    console.log(i); // Each iteration has its own i
}
console.log(i); // Error: i is not defined

Scope – Lexical (Static)

Functions can access variables from their outer scope.

function outer() {
    const outerVar = 'I am outside';
    
    function inner() {
        console.log(outerVar); // Accessible due to lexical scope
    }
    
    inner();
}

outer();

// Practical example
function createGreeting(greeting) {
    return function(name) {
        return `${greeting}, ${name}!`;
    };
}

const sayHello = createGreeting('Hello');
console.log(sayHello('Alice')); // Hello, Alice!

Arrow Function Challenge

Convert these functions to arrow functions and create a function that composes other functions.

// Convert to arrow functions
const multiply = (a, b) => a * b;

const createPerson = (name, age) => ({ name, age });

// Function composition
const compose = (f, g) => x => f(g(x));

const square = x => x * x;

Higher-Order Functions

Higher Order Functions – Map

map() creates a new array by transforming each element.

const numbers = [1, 2, 3, 4];
const doubled = numbers.map(num => num * 2);
console.log(doubled); // [2, 4, 6, 8]

// With named function
function toUpper(string) {
    return string.toUpperCase();
}

const names = ['alice', 'bob', 'charlie'];
const upperNames = names.map(toUpper);
console.log(upperNames); // ['ALICE', 'BOB', 'CHARLIE']

Higher Order Functions – Map Practice

Practical examples of using map().

// Transform array of objects
const products = [
    { id: 1, name: 'Laptop', price: 999 },
    { id: 2, name: 'Phone', price: 699 }
];

const productNames = products.map(product => product.name);
console.log(productNames); // ['Laptop', 'Phone']

// Add tax to prices
const withTax = products.map(product => ({
    ...product,
    priceWithTax: product.price * 1.2
}));

// Convert temperatures
const celsius = [0, 25, 100];
const fahrenheit = celsius.map(c => (c * 9/5) + 32);

Higher Order Functions – Filter

filter() creates a new array with elements that pass a test.

const numbers = [1, 2, 3, 4, 5, 6];
const evens = numbers.filter(num => num % 2 === 0);
console.log(evens); // [2, 4, 6]

// Filtering objects
const users = [
    { name: 'Alice', age: 25 },
    { name: 'Bob', age: 17 },
    { name: 'Charlie', age: 30 }
];

const adults = users.filter(user => user.age >= 18);
console.log(adults); // Alice and Charlie

Higher Order Functions – Filter Practice

Practical examples of using filter().

// Filter unique values
const values = [1, 2, 2, 3, 4, 4, 5];
const unique = values.filter((value, index, arr) => arr.indexOf(value) === index);

// Search functionality
function searchProducts(products, query) {
    return products.filter(product => 
        product.name.toLowerCase().includes(query.toLowerCase())
    );
}

// Active users
const allUsers = [
    { id: 1, name: 'Alice', isActive: true },
    { id: 2, name: 'Bob', isActive: false }
];
const activeUsers = allUsers.filter(user => user.isActive);

Higher Order Functions – Reduce

reduce() applies a function to accumulate a single result.

const numbers = [1, 2, 3, 4];
const sum = numbers.reduce((total, num) => total + num, 0);
console.log(sum); // 10

// Flatten array
const nested = [[1, 2], [3, 4], [5, 6]];
const flat = nested.reduce((flatArray, subArray) => flatArray.concat(subArray), []);

// Count occurrences
const fruits = ['apple', 'banana', 'apple', 'orange'];
const count = fruits.reduce((tally, fruit) => {
    tally[fruit] = (tally[fruit] || 0) + 1;
    return tally;
}, {});

Higher Order Functions – Reduce Practice

Practical examples of using reduce().

// Grouping by property
const people = [
    { name: 'Alice', age: 25 },
    { name: 'Bob', age: 30 },
    { name: 'Charlie', age: 25 }
];

const groupedByAge = people.reduce((groups, person) => {
    const key = person.age;
    if (!groups[key]) groups[key] = [];
    groups[key].push(person);
    return groups;
}, {});

// Calculate average
const grades = [85, 90, 78, 92, 88];
const average = grades.reduce((sum, grade) => sum + grade, 0) / grades.length;

// Compose functions
const compose = (...fns) => x => fns.reduceRight((acc, fn) => fn(acc), x);

Higher Order Functions – ForEach & Practice

forEach() executes a function for each array element.

const numbers = [1, 2, 3];
numbers.forEach(num => console.log(num * 2));

// Modifying DOM elements
const buttons = document.querySelectorAll('button');
buttons.forEach(button => {
    button.addEventListener('click', () => {
        console.log('Button clicked');
    });
});

// Side effects
const cart = [];
const products = [{ id: 1, name: 'Shirt' }, { id: 2, name: 'Pants' }];
products.forEach(product => cart.push(product));

Higher-Order Function Challenges

Higher Order Functions – Challenge

Create functions that:

  1. Process sales data to get total revenue by category
  2. Implement a custom filter function without using Array.filter
  3. Create a pipeline of functions that process data step-by-step
// 1. Total revenue by category
const sales = [
    { id: 1, category: 'Electronics', amount: 100 },
    { id: 2, category: 'Clothing', amount: 50 },
    { id: 3, category: 'Electronics', amount: 200 }
];

function getRevenueByCategory(transactions) {
    return transactions.reduce((totals, sale) => {
        totals[sale.category] = (totals[sale.category] || 0) + sale.amount;
        return totals;
    }, {});
}

// 2. Custom filter
function myFilter(array, callback) {
    const result = [];
    for (let item of array) {
        if (callback(item)) result.push(item);
    }
    return result;
}

// 3. Function pipeline
function createPipeline(...functions) {
    return function(input) {
        return functions.reduce((value, fn) => fn(value), input);
    };
}

const clean = str => str.trim();
const capitalize = str => str.charAt(0).toUpperCase() + str.slice(1);
const addGreeting = str => `Hello, ${str}!`;

const formatName = createPipeline(clean, capitalize, addGreeting);
console.log(formatName('  alice  ')); // Hello, Alice!
